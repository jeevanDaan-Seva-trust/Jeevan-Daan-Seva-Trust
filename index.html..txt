      /* Copy UPI ID / account number / IFSC */
      (function () {
        var status = document.getElementById('copy-status');

        function fallbackCopy(text) {
          var input = document.createElement('textarea');
          input.value = text;
          input.setAttribute('readonly', '');
          input.style.position = 'fixed';
          input.style.opacity = '0';
          document.body.appendChild(input);
          input.select();
          var ok = false;
          try { ok = document.execCommand('copy'); } catch (e) { ok = false; }
          document.body.removeChild(input);
          return ok;
        }

        function confirmCopy(button, label) {
          var copyIcon = button.querySelector('.icon-copy');
          var checkIcon = button.querySelector('.icon-check');
          if (copyIcon && checkIcon) {
            copyIcon.classList.add('hidden');
            checkIcon.classList.remove('hidden');
            window.setTimeout(function () {
              copyIcon.classList.remove('hidden');
              checkIcon.classList.add('hidden');
            }, 1800);
          }
          if (status) status.textContent = label + ' copy ho gaya';
        }

        document.querySelectorAll('.copy-btn').forEach(function (button) {
          button.addEventListener('click', function () {
            var text = button.getAttribute('data-copy') || '';
            var label = button.getAttribute('data-label') || 'Detail';
            if (navigator.clipboard && window.isSecureContext) {
              navigator.clipboard.writeText(text).then(
                function () { confirmCopy(button, label); },
                function () { if (fallbackCopy(text)) confirmCopy(button, label); }
              );
            } else if (fallbackCopy(text)) {
              confirmCopy(button, label);
            }
          });
        });
      })();

      /* Gallery lightbox */
      (function () {
        var box = document.getElementById('lightbox');
        var image = document.getElementById('lightbox-img');
        var caption = document.getElementById('lightbox-caption');
        var closeBtn = document.getElementById('lightbox-close');
        var lastFocused = null;
        if (!box || !image) return;

        function open(src, text, alt) {
          lastFocused = document.activeElement;
          image.src = '/.netlify/images?url=' + encodeURIComponent(src) + '&w=900&fm=webp';
          image.alt = alt || text || '';
          caption.textContent = text || '';
          box.classList.remove('hidden');
          box.classList.add('flex');
          document.body.style.overflow = 'hidden';
          closeBtn.focus();
        }

        function close() {
          box.classList.add('hidden');
          box.classList.remove('flex');
          document.body.style.overflow = '';
          image.src = '';
          if (lastFocused && lastFocused.focus) lastFocused.focus();
        }

        document.querySelectorAll('[data-lightbox]').forEach(function (trigger) {
          trigger.addEventListener('click', function () {
            var inner = trigger.querySelector('img');
            open(

              trigger.getAttribute('data-caption'),
              inner ? inner.getAttribute('alt') : ''
            );
          });
        });

          if (event.target === box) close();
        });
        document.addEventListener('keydown', function (event) {
          if (event.key === 'Escape' && !box.classList.contains('hidden')) close();
        });
      })();

      /* Reveal cards and photos as they scroll into view */
      (function () {
        var items = document.querySelectorAll('.reveal');
        if (!('IntersectionObserver' in window)) {
          items.forEach(function (item) { item.classList.add('is-visible'); });
          return;
        }
        var observer = new IntersectionObserver(function (entries) {
          entries.forEach(function (entry) {
            if (entry.isIntersecting) {
              entry.target.classList.add('is-visible');
              observer.unobserve(entry.target);
            }
          });
        }, { rootMargin: '0px 0px -40px 0px' });
        items.forEach(function (item) { observer.observe(item); });
      })();
